**Goal:** 实现元铸工坊 v0.1 的本体层核心（对象/链接/动作 DSL + staged 状态机）和 MCP 暴露，验证 H7 轻量本体柱。

**Architecture:** 
- 采用 FastAPI + SQLite 实现，模块化单体设计，支持单机模式 all-in-one 打包
- 本体层提供对象/链接/动作三件套 schema 和 staged writes 状态机
- MCP 通过 FastAPI 挂载 Streamable HTTP 端点，自动暴露对象查询和动作执行工具

**Tech Stack:** Python 3.11+, FastAPI, SQLite (aiosqlite), Pydantic v2, pytest, LiteLLM (预留接口)

**Spec:** `docs/specs/v0.1-design.md` 第 3.1 节（本体层）+ 第 3.2 节（MCP 暴露）+ 第 6 节里程碑 1-2

## Global Constraints

- **Python 版本**：3.11+（必需，typing 改进）
- **数据库**：SQLite（v0.1），PostgreSQL 预留迁移路径
- **API 协议**：REST + MCP Streamable HTTP
- **时间处理**：时间权威源用 PowerShell（Git Bash date 是 UTC），数据库时间用 UTC
- **测试策略**：TDD（pytest），每个 Task 必须包含测试
- **命名规范**：snake_case（Python），kebab-case（API 端点），PascalCase（类名）
- **文件编码**：UTF-8（所有源文件）
- **提交粒度**：每个 Task 一个 commit，commit message 用中文 feat: 前缀
- **staged writes**：所有写操作默认 staged 人审（ADR-006）
- **幂等性**：所有动作执行必须支持幂等键
- **补偿事务**：每个动作必须实现 revert 逻辑

from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession, async_sessionmaker
from sqlalchemy.orm import declarative_base
from contextlib import asynccontextmanager
from yuanzhu.config import settings

Base = declarative_base()

# 创建异步引擎
engine = create_async_engine(
    settings.database_url,
    echo=settings.debug,
    future=True,
)

# 创建异步会话工厂
async_session = async_sessionmaker(
    engine,
    class_=AsyncSession,
    expire_on_commit=False,
)


@asynccontextmanager
async def get_db():
    """获取数据库会话的异步上下文管理器"""
    async with async_session() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()


async def init_db():
    """初始化数据库（创建表）"""
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
```

from sqlalchemy import Column, Integer, String, Text, Boolean, DateTime, ForeignKey, JSON
from sqlalchemy.orm import relationship
from datetime import datetime, timezone
from yuanzhu.db.database import Base


def utcnow():
    """返回 UTC 时间（避免 Git Bash date 是 UTC 的混淆）"""
    return datetime.now(timezone.utc)


class ObjectType(Base):
    """对象类型定义"""
    __tablename__ = "object_type"
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    domain = Column(String(100), nullable=False, index=True)  # 领域（如 aiqa, workflow）
    name = Column(String(100), nullable=False, index=True)  # 类型名（如 Bug, TestCase）
    title_key = Column(String(100), nullable=True)  # 标题属性名（用于显示）
    description = Column(Text, nullable=True)
    schema_json = Column(JSON, nullable=False)  # JSON Schema 定义属性
    exposed = Column(Boolean, default=False)  # 是否通过 MCP 暴露
    created_at = Column(DateTime, default=utcnow, nullable=False)
    
    # 关系
    objects = relationship("Object", back_populates="object_type")
    
    __table_args__ = (
        {"comment": "对象类型定义表"},
    )


class Object(Base):
    """对象实例"""
    __tablename__ = "object"
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    type_id = Column(Integer, ForeignKey("object_type.id"), nullable=False, index=True)
    properties_json = Column(JSON, nullable=False, default=dict)  # 对象属性
    title = Column(String(500), nullable=True)  # 标题（从 title_key 自动提取）
    created_by = Column(String(100), nullable=True, index=True)  # 创建者（agent 或用户）
    created_at = Column(DateTime, default=utcnow, nullable=False)
    updated_at = Column(DateTime, default=utcnow, onupdate=utcnow, nullable=False)
    
    # 关系
    object_type = relationship("ObjectType", back_populates="objects")
    
    __table_args__ = (
        {"comment": "对象实例表"},
    )


class LinkType(Base):
    """链接类型定义"""
    __tablename__ = "link_type"
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    domain = Column(String(100), nullable=False, index=True)
    name = Column(String(100), nullable=False, index=True)  # 链接名（如 has_bug, parent_of）
    source_type_id = Column(Integer, ForeignKey("object_type.id"), nullable=False)
    target_type_id = Column(Integer, ForeignKey("object_type.id"), nullable=False)
    cardinality = Column(String(50), nullable=False, default="many")  # one | many
    
    __table_args__ = (
        {"comment": "链接类型定义表"},
    )


class Link(Base):
    """链接实例"""
    __tablename__ = "link"
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    type_id = Column(Integer, ForeignKey("link_type.id"), nullable=False, index=True)
    source_id = Column(Integer, ForeignKey("object.id"), nullable=False, index=True)
    target_id = Column(Integer, ForeignKey("object.id"), nullable=False, index=True)
    metadata_json = Column(JSON, nullable=True)  # 链接元数据
    
    __table_args__ = (
        {"comment": "链接实例表"},
    )


class ActionType(Base):
    """动作类型定义"""
    __tablename__ = "action_type"
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    domain = Column(String(100), nullable=False, index=True)
    name = Column(String(100), nullable=False, index=True)
    
    # 参数 schema（JSON Schema）
    params_schema_json = Column(JSON, nullable=False, default=dict)
    
    # 运行时校验（如 status=Open 才可改优先级）
    submission_criteria_json = Column(JSON, nullable=True)
    
    # 副作用列表 [{type: webhook|notify, ...}]
    side_effects_json = Column(JSON, nullable=True)
    
    # 自主性等级（ADR-006: L1-L5，读高写低）
    autonomy_level = Column(Integer, default=2, nullable=False)  # 默认 L2（staged 人审）
    
    # 是否需要 staged 审批
    requires_staging = Column(Boolean, default=True, nullable=False)
    
    # agent 工具描述（Palantir Agent tool description）
    description_for_agent = Column(Text, nullable=True)
    
    # 幂等键模板（如 "{object_id}-{action}"）
    idempotency_key_template = Column(String(200), nullable=True)
    
    created_at = Column(DateTime, default=utcnow, nullable=False)
    
    __table_args__ = (
        {"comment": "动作类型定义表"},
    )


class ActionExec(Base):
    """动作执行记录"""
    __tablename__ = "action_exec"
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    action_type_id = Column(Integer, ForeignKey("action_type.id"), nullable=False, index=True)
    
    # 幂等键（防重）
    idempotency_key = Column(String(200), nullable=True, unique=True, index=True)
    
    # 参数
    params_json = Column(JSON, nullable=False, default=dict)
    
    # 状态：staged | approved | applied | rejected | reverted
    status = Column(String(50), nullable=False, default="staged", index=True)
    
    # 审批信息
    staged_by = Column(String(100), nullable=True, index=True)  # 谁 staged
    staged_at = Column(DateTime, default=utcnow, nullable=False)
    
    reviewed_by = Column(String(100), nullable=True)  # 审批人
    reviewed_at = Column(DateTime, nullable=True)
    review_comment = Column(Text, nullable=True)  # 审批意见
    
    applied_at = Column(DateTime, nullable=True)  # 应用时间
    reverted_at = Column(DateTime, nullable=True)  # 撤销时间
    
    # 执行日志
    exec_log_json = Column(JSON, nullable=True)
    
    __table_args__ = (
        {"comment": "动作执行记录表"},
    )
```

def parse_object_type_yaml(yaml_content: str) -> Dict[str, Any]:
    """
    解析对象类型 DSL（YAML 格式）
    
    输入格式：
    ```yaml
    name: Bug
    domain: aiqa
    title_key: title
    description: Bug 类型描述
    exposed: true
    properties:
      title:
        type: string
        required: true
      status:
        type: string
        enum: [Open, Closed]
        default: Open
    ```
    
    输出：符合 ObjectTypeCreate 的字典
    """
    data = yaml.safe_load(yaml_content)
    
    if not data:
        raise ValueError("YAML 内容为空")
    
    # 提取基本字段
    name = data.get("name")
    domain = data.get("domain")
    title_key = data.get("title_key")
    description = data.get("description")
    exposed = data.get("exposed", False)
    
    if not name:
        raise ValueError("缺少必需字段: name")
    if not domain:
        raise ValueError("缺少必需字段: domain")
    
    # 解析 properties 为 JSON Schema
    properties_data = data.get("properties", {})
    properties = {}
    required = []
    
    for prop_name, prop_def in properties_data.items():
        prop_schema = ObjectPropertySchema(
            type=prop_def.get("type", "string"),
            required=prop_def.get("required", False),
            enum=prop_def.get("enum"),
            default=prop_def.get("default"),
            minimum=prop_def.get("minimum"),
            maximum=prop_def.get("maximum"),
            description=prop_def.get("description"),
        )
        properties[prop_name] = prop_schema
        
        if prop_schema.required:
            required.append(prop_name)
    
    # 构建 JSON Schema
    schema_json = {
        "type": "object",
        "properties": {},
        "required": required,
    }
    
    for prop_name, prop_schema in properties.items():
        prop_json = {"type": prop_schema.type}
        
        if prop_schema.enum:
            prop_json["enum"] = prop_schema.enum
        if prop_schema.default is not None:
            prop_json["default"] = prop_schema.default
        if prop_schema.minimum is not None:
            prop_json["minimum"] = prop_schema.minimum
        if prop_schema.maximum is not None:
            prop_json["maximum"] = prop_schema.maximum
        if prop_schema.description:
            prop_json["description"] = prop_schema.description
        
        schema_json["properties"][prop_name] = prop_json
    
    # 验证生成的 schema
    Draft7Validator.check_schema(schema_json)
    
    return {
        "name": name,
        "domain": domain,
        "title_key": title_key,
        "description": description,
        "schema_json": schema_json,
        "exposed": exposed,
    }

from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from typing import List, Optional, Dict, Any
from yuanzhu.db.models import ObjectType, Object
from yuanzhu.models.object import ObjectTypeCreate, ObjectCreate, ObjectUpdate
from jsonschema import validate, ValidationError


class ObjectStore:
    """对象存储层"""
    
    def __init__(self, session: AsyncSession):
        self.session = session
    
    async def create_type(self, type_data: ObjectTypeCreate) -> ObjectType:
        """创建对象类型"""
        
        # 检查是否已存在
        existing = await self.get_type_by_name(type_data.domain, type_data.name)
        if existing:
            # 更新现有类型
            existing.title_key = type_data.title_key
            existing.description = type_data.description
            existing.schema_json = self._build_schema(type_data.properties)
            existing.exposed = type_data.exposed
            await self.session.flush()
            return existing
        
        # 创建新类型
        obj_type = ObjectType(
            name=type_data.name,
            domain=type_data.domain,
            title_key=type_data.title_key,
            description=type_data.description,
            schema_json=self._build_schema(type_data.properties),
            exposed=type_data.exposed,
        )
        
        self.session.add(obj_type)
        await self.session.flush()
        await self.session.refresh(obj_type)
        
        return obj_type
    
    async def get_type(self, type_id: int) -> Optional[ObjectType]:
        """获取对象类型"""
        result = await self.session.execute(
            select(ObjectType).where(ObjectType.id == type_id)
        )
        return result.scalar_one_or_none()
    
    async def get_type_by_name(self, domain: str, name: str) -> Optional[ObjectType]:
        """按名称获取对象类型"""
        result = await self.session.execute(
            select(ObjectType).where(
                ObjectType.domain == domain,
                ObjectType.name == name,
            )
        )
        return result.scalar_one_or_none()
    
    async def create_object(self, obj_data: ObjectCreate) -> Object:
        """创建对象实例"""
        
        # 获取对象类型
        obj_type = await self.get_type(obj_data.type_id)
        if not obj_type:
            raise ValueError(f"对象类型不存在: {obj_data.type_id}")
        
        # 验证属性是否符合 schema
        self._validate_properties(obj_type, obj_data.properties)
        
        # 提取标题
        title = obj_data.title
        if not title and obj_type.title_key:
            title = obj_data.properties.get(obj_type.title_key)
        
        # 创建对象
        obj = Object(
            type_id=obj_data.type_id,
            properties=obj_data.properties,
            title=title,
            created_by=obj_data.created_by,
        )
        
        self.session.add(obj)
        await self.session.flush()
        await self.session.refresh(obj)
        
        return obj
    
    async def get_object(self, obj_id: int) -> Optional[Object]:
        """获取对象实例"""
        result = await self.session.execute(
            select(Object).where(Object.id == obj_id)
        )
        return result.scalar_one_or_none()
    
    async def list_objects(
        self,
        type_id: Optional[int] = None,
        domain: Optional[str] = None,
        created_by: Optional[str] = None,
        limit: int = 100,
        offset: int = 0,
    ) -> List[Object]:
        """列出对象"""
        query = select(Object)
        
        if type_id:
            query = query.where(Object.type_id == type_id)
        if created_by:
            query = query.where(Object.created_by == created_by)
        if domain:
            query = query.join(ObjectType).where(ObjectType.domain == domain)
        
        query = query.order_by(Object.created_at.desc()).limit(limit).offset(offset)
        
        result = await self.session.execute(query)
        return list(result.scalars().all())
    
    async def update_object(self, obj_id: int, update_data: ObjectUpdate) -> Object:
        """更新对象实例"""
        
        obj = await self.get_object(obj_id)
        if not obj:
            raise ValueError(f"对象不存在: {obj_id}")
        
        # 合并属性
        if update_data.properties:
            new_properties = {**obj.properties, **update_data.properties}
            
            # 验证新属性
            obj_type = await self.get_type(obj.type_id)
            self._validate_properties(obj_type, new_properties)
            
            obj.properties = new_properties
        
        # 更新标题
        if update_data.title:
            obj.title = update_data.title
        elif obj.properties:
            # 重新提取标题
            obj_type = await self.get_type(obj.type_id)
            if obj_type and obj_type.title_key:
                obj.title = obj.properties.get(obj_type.title_key)
        
        await self.session.flush()
        await self.session.refresh(obj)
        
        return obj
    
    async def delete_object(self, obj_id: int) -> bool:
        """删除对象实例"""
        obj = await self.get_object(obj_id)
        if not obj:
            return False
        
        await self.session.delete(obj)
        await self.session.flush()
        
        return True
    
    def _build_schema(self, properties: Dict[str, Any]) -> Dict[str, Any]:
        """构建 JSON Schema"""
        schema = {
            "type": "object",
            "properties": {},
            "required": [],
        }
        
        for prop_name, prop_def in properties.items():
            prop_json = {"type": prop_def.get("type", "string")}
            
            if "enum" in prop_def:
                prop_json["enum"] = prop_def["enum"]
            if "default" in prop_def:
                prop_json["default"] = prop_def["default"]
            if "minimum" in prop_def:
                prop_json["minimum"] = prop_def["minimum"]
            if "maximum" in prop_def:
                prop_json["maximum"] = prop_def["maximum"]
            if "description" in prop_def:
                prop_json["description"] = prop_def["description"]
            
            schema["properties"][prop_name] = prop_json
            
            if prop_def.get("required", False):
                schema["required"].append(prop_name)
        
        return schema
    
    def _validate_properties(self, obj_type: ObjectType, properties: Dict[str, Any]):
        """验证属性是否符合 schema"""
        try:
            validate(instance=properties, schema=obj_type.schema_json)
        except ValidationError as e:
            raise ValueError(f"属性验证失败: {e.message}")
```

```python
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from typing import Optional, Dict, Any
from datetime import datetime, timezone
from yuanzhu.db.models import ActionType, ActionExec, Object
from yuanzhu.ontology.object_store import ObjectStore
from jsonschema import validate, ValidationError


def utcnow():
    """返回 UTC 时间"""
    return datetime.now(timezone.utc)


class StagedStateMachine:
    """Staged writes 状态机"""
    
    def __init__(self, session: AsyncSession):
        self.session = session
        self.object_store = ObjectStore(session)
    
    async def stage(
        self,
        action_type_id: int,
        params: Dict[str, Any],
        staged_by: str,
        idempotency_key: Optional[str] = None,
    ) -> ActionExec:
        """Stage 动作（暂存待审）"""
        
        # 获取动作类型
        action_type = await self.session.execute(
            select(ActionType).where(ActionType.id == action_type_id)
        )
        action_type = action_type.scalar_one_or_none()
        
        if not action_type:
            raise ValueError(f"动作类型不存在: {action_type_id}")
        
        # 验证参数
        try:
            validate(instance=params, schema=action_type.params_schema_json)
        except ValidationError as e:
            raise ValueError(f"参数验证失败: {e.message}")
        
        # 幂等性检查
        if idempotency_key:
            existing = await self.session.execute(
                select(ActionExec).where(ActionExec.idempotency_key == idempotency_key)
            )
            existing_exec = existing.scalar_one_or_none()
            if existing_exec:
                # 返回已存在的记录（幂等）
                return existing_exec
        
        # 创建执行记录
        exec = ActionExec(
            action_type_id=action_type_id,
            params_json=params,
            idempotency_key=idempotency_key,
            status="staged",
            staged_by=staged_by,
            staged_at=utcnow(),
        )
        
        self.session.add(exec)
        await self.session.flush()
        await self.session.refresh(exec)
        
        return exec
    
    async def approve(
        self,
        exec_id: int,
        reviewed_by: str,
        review_comment: Optional[str] = None,
    ) -> ActionExec:
        """批准 staged 动作"""
        
        exec = await self._get_exec(exec_id)
        
        if exec.status != "staged":
            raise ValueError(f"动作状态不是 staged: {exec.status}")
        
        exec.status = "approved"
        exec.reviewed_by = reviewed_by
        exec.reviewed_at = utcnow()
        exec.review_comment = review_comment
        
        await self.session.flush()
        await self.session.refresh(exec)
        
        return exec
    
    async def apply(self, exec_id: int) -> ActionExec:
        """应用 approved 动作"""
        
        exec = await self._get_exec(exec_id)
        
        if exec.status != "approved":
            raise ValueError(f"动作状态不是 approved: {exec.status}")
        
        # 执行动作（这里简化为直接更新对象属性）
        # 实际项目中应该根据 action_type 和 params 执行具体逻辑
        await self._execute_action(exec)
        
        exec.status = "applied"
        exec.applied_at = utcnow()
        
        await self.session.flush()
        await self.session.refresh(exec)
        
        return exec
    
    async def reject(
        self,
        exec_id: int,
        reviewed_by: str,
        review_comment: str,
    ) -> ActionExec:
        """拒绝 staged 动作"""
        
        exec = await self._get_exec(exec_id)
        
        if exec.status != "staged":
            raise ValueError(f"动作状态不是 staged: {exec.status}")
        
        exec.status = "rejected"
        exec.reviewed_by = reviewed_by
        exec.reviewed_at = utcnow()
        exec.review_comment = review_comment
        
        await self.session.flush()
        await self.session.refresh(exec)
        
        return exec
    
    async def revert(self, exec_id: int) -> ActionExec:
        """撤销已应用动作（补偿事务）"""
        
        exec = await self._get_exec(exec_id)
        
        if exec.status != "applied":
            raise ValueError(f"动作状态不是 applied: {exec.status}")
        
        # 执行补偿（这里简化为恢复对象属性）
        await self._revert_action(exec)
        
        exec.status = "reverted"
        exec.reverted_at = utcnow()
        
        await self.session.flush()
        await self.session.refresh(exec)
        
        return exec
    
    async def _get_exec(self, exec_id: int) -> ActionExec:
        """获取动作执行记录"""
        result = await self.session.execute(
            select(ActionExec).where(ActionExec.id == exec_id)
        )
        exec = result.scalar_one_or_none()
        
        if not exec:
            raise ValueError(f"动作执行记录不存在: {exec_id}")
        
        return exec
    
    async def _execute_action(self, exec: ActionExec):
        """执行动作的具体逻辑（示例：更新对象属性）"""
        
        action_type = await self.session.execute(
            select(ActionType).where(ActionType.id == exec.action_type_id)
        )
        action_type = action_type.scalar_one_or_none()
        
        # 示例：ChangePriority 动作
        if action_type and action_type.name == "ChangePriority":
            object_id = exec.params_json.get("object_id")
            priority = exec.params_json.get("priority")
            
            if object_id and priority is not None:
                obj = await self.object_store.get_object(object_id)
                if obj:
                    # 保存原始值（用于 revert）
                    if not exec.exec_log_json:
                        exec.exec_log_json = {
                            "original_priority": obj.properties.get("priority"),
                        }
                    
                    # 更新对象
                    obj.properties["priority"] = priority
                    await self.session.flush()
    
    async def _revert_action(self, exec: ActionExec):
        """执行补偿事务（恢复原始状态）"""
        
        action_type = await self.session.execute(
            select(ActionType).where(ActionType.id == exec.action_type_id)
        )
        action_type = action_type.scalar_one_or_none()
        
        # 示例：ChangePriority 动作的补偿
        if action_type and action_type.name == "ChangePriority":
            object_id = exec.params_json.get("object_id")
            
            if object_id and exec.exec_log_json:
                obj = await self.object_store.get_object(object_id)
                if obj:
                    # 恢复原始值
                    original_priority = exec.exec_log_json.get("original_priority")
                    if original_priority is not None:
                        obj.properties["priority"] = original_priority
                        await self.session.flush()
```

        obj_type = result.scalar_one_or_none()
        
        if not obj_type or not obj_type.exposed:
            return None
        
        # 工具名：query_<domain>_<name>
        tool_name = f"query_{obj_type.domain}_{obj_type.name}".lower()
        
        # 构建输入 schema（查询条件）
        input_schema = {
            "type": "object",
            "properties": {
                "filter": {
                    "type": "object",
                    "description": f"过滤条件（{obj_type.name} 属性）",
                },
                "limit": {
                    "type": "integer",
                    "description": "返回数量限制",
                    "default": 10,
                },
            },
        }
        
        return {
            "name": tool_name,
            "description": f"查询 {obj_type.description or obj_type.name} 对象",
            "inputSchema": input_schema,
        }
    
    async def generate_action_tool(self, action_type_id: int) -> Optional[Dict[str, Any]]:
        """生成执行动作 MCP 工具"""
        
        result = await self.session.execute(
            select(ActionType).where(ActionType.id == action_type_id)
        )
        action_type = result.scalar_one_or_none()
        
        if not action_type:
            return None
        
        # 工具名：execute_<domain>_<name>
        tool_name = f"execute_{action_type.domain}_{action_type.name}".lower()
        
        # 使用动作类型的参数 schema
        input_schema = action_type.params_schema_json
        
        # 如果需要 staged，在描述中说明
        description = action_type.description_for_agent or f"执行 {action_type.name} 动作"
        if action_type.requires_staging:
            description += "（需要审批）"
        
        return {
            "name": tool_name,
            "description": description,
            "inputSchema": input_schema,
        }
    
    async def list_tools(self) -> List[Dict[str, Any]]:
        """列出所有暴露的工具"""
        tools = []
        
        # 查询所有暴露的对象类型
        result = await self.session.execute(
            select(ObjectType).where(ObjectType.exposed == True)
        )
        object_types = result.scalars().all()
        
        for obj_type in object_types:
            tool = await self.generate_query_tool(obj_type.id)
            if tool:

        # 查询所有动作类型
        result = await self.session.execute(select(ActionType))
        action_types = result.scalars().all()
        
        for action_type in action_types:
            tool = await self.generate_action_tool(action_type.id)
            if tool:
                tools.append(tool)
        
        # 添加审批相关工具
        tools.extend(self._get_approval_tools())

from sqlalchemy.ext.asyncio import AsyncSession
from typing import Dict, Any, Optional
from yuanzhu.ontology.object_store import ObjectStore
from yuanzhu.ontology.action_store import ActionStore
from yuanzhu.staged.state_machine import StagedStateMachine
from yuanzhu.db.models import ActionExec
from sqlalchemy import select


class MCPHandlers:
    """MCP 请求处理器"""
    
    def __init__(self, session: AsyncSession):
        self.session = session
        self.object_store = ObjectStore(session)
        self.action_store = ActionStore(session)
        self.state_machine = StagedStateMachine(session)
    
    async def handle_query_objects(
        self,
        tool_name: str,
        params: Dict[str, Any],
    ) -> Dict[str, Any]:
        """处理查询对象请求"""
        
        # 解析工具名：query_<domain>_<name>
        parts = tool_name.split("_")
        if len(parts) < 3:
            return {"error": "无效的工具名"}
        
        domain = parts[1]
        type_name = "_".join(parts[2:])
        
        # 查询对象
        filters = params.get("filter", {})
        limit = params.get("limit", 10)
        
        objects = await self.object_store.list_objects(
            domain=domain,
            limit=limit,
        )
        
        # 应用过滤条件
        if filters:
            filtered = []
            for obj in objects:
                match = all(
                    obj.properties.get(k) == v
                    for k, v in filters.items()
                )
                if match:
                    filtered.append(obj)
            objects = filtered
        
        return {
            "objects": [
                {
                    "id": obj.id,
                    "properties": obj.properties,
                    "title": obj.title,
                }
                for obj in objects
            ],
        }
    
    async def handle_execute_action(
        self,
        tool_name: str,
        params: Dict[str, Any],
        agent_id: str,
    ) -> Dict[str, Any]:
        """处理执行动作请求"""
        
        # 解析工具名：execute_<domain>_<name>
        parts = tool_name.split("_")
        if len(parts) < 3:
            return {"error": "无效的工具名"}
        
        domain = parts[1]
        action_name = "_".join(parts[2:])
        
        # 获取动作类型
        action_type = await self.action_store.get_type_by_name(domain, action_name)
        if not action_type:
            return {"error": f"动作类型不存在: {domain}/{action_name}"}
        
        # 生成幂等键
        idempotency_key = None
        if action_type.idempotency_key_template:
            idempotency_key = action_type.idempotency_key_template.format(
                **params,
                name=action_name,
            )
        
        # Stage 动作
        exec = await self.state_machine.stage(
            action_type_id=action_type.id,
            params=params,
            staged_by=agent_id,
            idempotency_key=idempotency_key,
        )
        
        result = {
            "exec_id": exec.id,
            "status": exec.status,
        }
        
        # 如果不需要 staged，直接批准和应用
        if not action_type.requires_staging:
            await self.state_machine.approve(exec.id, reviewed_by="system")
            await self.state_machine.apply(exec.id)
            result["status"] = "applied"
            result["message"] = "动作已执行"
        else:
            result["message"] = "动作已暂存，等待审批"
        
        return result
    
    async def handle_list_pending_approvals(
        self,
        params: Dict[str, Any],
    ) -> Dict[str, Any]:
        """处理列出待审批请求"""
        
        domain = params.get("domain")
        
        query = select(ActionExec).where(ActionExec.status == "staged")
        
        result = await self.session.execute(query)
        pending = result.scalars().all()
        
        return {
            "pending": [
                {
                    "exec_id": exec.id,
                    "action_type_id": exec.action_type_id,
                    "params": exec.params_json,
                    "staged_by": exec.staged_by,
                    "staged_at": exec.staged_at.isoformat(),
                }
                for exec in pending
            ],
        }
    
    async def handle_approve_action(
        self,
        params: Dict[str, Any],
        reviewer_id: str,
    ) -> Dict[str, Any]:
        """处理批准动作请求"""
        
        exec_id = params.get("exec_id")
        comment = params.get("comment")
        
        if not exec_id:
            return {"error": "缺少 exec_id"}
        
        try:
            exec = await self.state_machine.approve(
                exec_id,
                reviewed_by=reviewer_id,
                review_comment=comment,
            )
            
            # 自动应用
            await self.state_machine.apply(exec_id)
            
            return {
                "exec_id": exec.id,
                "status": "applied",
                "message": "动作已批准并执行",
            }
        except Exception as e:
            return {"error": str(e)}
    
    async def handle_reject_action(
        self,
        params: Dict[str, Any],
        reviewer_id: str,
    ) -> Dict[str, Any]:
        """处理拒绝动作请求"""
        
        exec_id = params.get("exec_id")
        comment = params.get("comment")
        
        if not exec_id:
            return {"error": "缺少 exec_id"}
        if not comment:
            return {"error": "缺少拒绝理由"}
        
        try:
            exec = await self.state_machine.reject(
                exec_id,
                reviewed_by=reviewer_id,
                review_comment=comment,
            )
            
            return {
                "exec_id": exec.id,
                "status": "rejected",
                "message": "动作已拒绝",
            }
        except Exception as e:
            return {"error": str(e)}

```python
from fastapi import FastAPI, Request, Response
from fastapi.responses import JSONResponse
from typing import Dict, Any
import json
from yuanzhu.db.database import get_db
from yuanzhu.mcp.tools import MCPToolGenerator
from yuanzhu.mcp.handlers import MCPHandlers


class MCPServer:
    """MCP Streamable HTTP 服务端"""
    
    def __init__(self, app: FastAPI, endpoint: str = "/mcp"):
        self.app = app
        self.endpoint = endpoint
        self._setup_routes()
    
    def _setup_routes(self):
        """设置 MCP 路由"""
        
        @self.app.post(self.endpoint)
        async def handle_mcp_request(request: Request):
            """处理 MCP 请求"""
            
            # 解析请求
            body = await request.json()
            method = body.get("method")
            params = body.get("params", {})
            
            # 获取 agent 身份（从 header 或 params）
            agent_id = request.headers.get("X-Agent-ID", "unknown")
            
            # 获取数据库会话
            async for session in get_db():
                try:
                    result = await self._dispatch(session, method, params, agent_id)
                    return JSONResponse(result)
                except Exception as e:
                    return JSONResponse(
                        {"error": str(e)},
                        status_code=500,
                    )
    
    async def _dispatch(
        self,
        session,
        method: str,
        params: Dict[str, Any],
        agent_id: str,
    ) -> Dict[str, Any]:
        """分发 MCP 请求"""
        
        if method == "tools/list":
            generator = MCPToolGenerator(session)
            tools = await generator.list_tools()
            return {"tools": tools}
        
        elif method == "tools/call":
            tool_name = params.get("name")
            tool_params = params.get("arguments", {})
            
            handlers = MCPHandlers(session)
            
            # 根据工具名路由
            if tool_name.startswith("query_"):
                return await handlers.handle_query_objects(tool_name, tool_params)
            elif tool_name.startswith("execute_"):
                return await handlers.handle_execute_action(tool_name, tool_params, agent_id)
            elif tool_name == "list_pending_approvals":
                return await handlers.handle_list_pending_approvals(tool_params)
            elif tool_name == "approve_action":
                return await handlers.handle_approve_action(tool_params, agent_id)
            elif tool_name == "reject_action":
                return await handlers.handle_reject_action(tool_params, agent_id)
            else:
                return {"error": f"未知工具: {tool_name}"}
        
        else:
            return {"error": f"未知方法: {method}"}
```

    state_machine = StagedStateMachine(db)
    
    try:
        exec = await state_machine.reject(
            exec_id,
            reviewed_by=reviewer_id,
            review_comment=review_comment,
        )
        return ActionExecResponse.from_orm(exec)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))


@router.post("/{exec_id}/revert", response_model=ActionExecResponse)
async def revert_action(
    exec_id: int,
    db: AsyncSession = Depends(get_db),
):
    """撤销已应用动作"""
    state_machine = StagedStateMachine(db)
    
    try:
        exec = await state_machine.revert(exec_id)
        return ActionExecResponse.from_orm(exec)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
```
