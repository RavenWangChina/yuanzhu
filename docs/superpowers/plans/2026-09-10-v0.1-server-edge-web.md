```python
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, update
from typing import List, Optional
from datetime import datetime, timezone
from yuanzhu.db.models import Node
from yuanzhu.node.models import NodeCreate, NodeHeartbeat


def utcnow():
    return datetime.now(timezone.utc)


class NodeStore:
    """节点存储层"""
    
    def __init__(self, session: AsyncSession):
        self.session = session
    
    async def register(self, node_data: NodeCreate) -> Node:
        """注册节点"""
        
        # 检查是否已存在
        existing = await self.get_by_name(node_data.name)
        if existing:
            # 更新现有节点
            existing.token = node_data.token
            existing.capabilities = node_data.capabilities
            existing.status = "online"
            existing.last_heartbeat = utcnow()
            await self.session.flush()
            return existing
        
        node = Node(
            name=node_data.name,
            token=node_data.token,
            capabilities=node_data.capabilities,
            status="online",
            last_heartbeat=utcnow(),
        )
        
        self.session.add(node)
        await self.session.flush()
        await self.session.refresh(node)
        
        return node
    
    async def heartbeat(self, node_id: int, heartbeat: NodeHeartbeat) -> Node:
        """更新节点心跳"""
        
        node = await self.get(node_id)
        if not node:
            raise ValueError(f"节点不存在: {node_id}")
        
        node.status = heartbeat.status
        node.current_task_id = heartbeat.current_task_id
        node.resource_usage = heartbeat.resource_usage
        node.last_heartbeat = utcnow()
        
        await self.session.flush()
        await self.session.refresh(node)
        
        return node
    
    async def get(self, node_id: int) -> Optional[Node]:
        """获取节点"""
        result = await self.session.execute(
            select(Node).where(Node.id == node_id)
        )
        return result.scalar_one_or_none()
    
    async def get_by_name(self, name: str) -> Optional[Node]:
        """按名称获取节点"""
        result = await self.session.execute(
            select(Node).where(Node.name == name)
        )
        return result.scalar_one_or_none()
    
    async def list(
        self,
        status: Optional[str] = None,
        limit: int = 100,
    ) -> List[Node]:
        """列出节点"""
        query = select(Node)
        
        if status:
            query = query.where(Node.status == status)
        
        query = query.order_by(Node.registered_at.desc()).limit(limit)
        
        result = await self.session.execute(query)
        return list(result.scalars().all())
    
    async def update_status(self, node_id: int, status: str) -> Node:
        """更新节点状态"""
        
        node = await self.get(node_id)
        if not node:
            raise ValueError(f"节点不存在: {node_id}")
        
        node.status = status
        await self.session.flush()
        await self.session.refresh(node)
        
        return node
```

```python
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from typing import Optional
from datetime import datetime, timezone
from yuanzhu.db.models import Task


def utcnow():
    return datetime.now(timezone.utc)


class TaskStateMachine:
    """任务状态机"""
    
    VALID_TRANSITIONS = {
        "pending": ["dispatched", "cancelled"],
        "dispatched": ["running", "cancelled"],
        "running": ["done", "failed", "cancelled"],
        "done": [],  # 终态
        "failed": [],  # 终态
        "cancelled": [],  # 终态
    }
    
    def __init__(self, session: AsyncSession):
        self.session = session
    
    async def transition(self, task_id: int, new_status: str) -> Task:
        """状态转换"""
        
        task = await self._get_task(task_id)
        
        # 验证转换合法性
        if new_status not in self.VALID_TRANSITIONS.get(task.status, []):
            raise ValueError(
                f"非法状态转换: {task.status} → {new_status}"
            )
        
        task.status = new_status
        
        # 终态时记录完成时间
        if new_status in ["done", "failed", "cancelled"]:
            task.completed_at = utcnow()
        
        await self.session.flush()
        await self.session.refresh(task)
        
        return task
    
    async def _get_task(self, task_id: int) -> Task:
        """获取任务"""
        result = await self.session.execute(
            select(Task).where(Task.id == task_id)
        )
        task = result.scalar_one_or_none()
        
        if not task:
            raise ValueError(f"任务不存在: {task_id}")
        
        return task
```

```python
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from typing import List, Optional
from datetime import datetime, timezone
from yuanzhu.db.models import Task, Node
from yuanzhu.task.models import TaskCreate
from yuanzhu.task.state_machine import TaskStateMachine


def utcnow():
    return datetime.now(timezone.utc)


class TaskOrchestrator:
    """任务编排器"""
    
    def __init__(self, session: AsyncSession):
        self.session = session
        self.state_machine = TaskStateMachine(session)
    
    async def create(self, task_data: TaskCreate) -> Task:
        """创建任务"""
        
        task = Task(
            template_id=task_data.template_id,
            template_version=task_data.template_version,
            params_json=task_data.params,
            status="pending",
            created_by=task_data.created_by,
        )
        
        self.session.add(task)
        await self.session.flush()
        await self.session.refresh(task)
        
        return task
    
    async def dispatch(self, task_id: int, node_id: int) -> Task:
        """分发任务到节点"""
        
        task = await self.state_machine._get_task(task_id)
        
        # 验证节点存在且在线
        result = await self.session.execute(
            select(Node).where(Node.id == node_id)
        )
        node = result.scalar_one_or_none()
        
        if not node:
            raise ValueError(f"节点不存在: {node_id}")
        
        if node.status != "online":
            raise ValueError(f"节点不在线: {node.status}")
        
        # 分配节点
        task.assigned_node_id = node_id
        
        # 状态转换
        await self.state_machine.transition(task_id, "dispatched")
        
        return task
    
    async def pull(self, node_id: int, limit: int = 10) -> List[Task]:
        """节点拉取任务"""
        
        # 查询分配给该节点且状态为 dispatched 的任务
        result = await self.session.execute(
            select(Task)
            .where(
                Task.assigned_node_id == node_id,
                Task.status == "dispatched",
            )
            .limit(limit)
        )
        
        tasks = list(result.scalars().all())
        
        # 标记为 running
        for task in tasks:
            await self.state_machine.transition(task.id, "running")
        
        return tasks
    
    async def update_status(
        self,
        task_id: int,
        status: str,
        result: Optional[dict] = None,
        exec_log: Optional[dict] = None,
    ) -> Task:
        """更新任务状态和结果"""
        
        task = await self.state_machine.transition(task_id, status)
        
        if result:
            task.result_json = result
        if exec_log:
            task.exec_log_json = exec_log
        
        await self.session.flush()
        await self.session.refresh(task)
        
        return task
    
    async def list(
        self,
        status: Optional[str] = None,
        node_id: Optional[int] = None,
        created_by: Optional[str] = None,
        limit: int = 100,
    ) -> List[Task]:
        """列出任务"""
        query = select(Task)
        
        if status:
            query = query.where(Task.status == status)
        if node_id:
            query = query.where(Task.assigned_node_id == node_id)
        if created_by:
            query = query.where(Task.created_by == created_by)
        
        query = query.order_by(Task.created_at.desc()).limit(limit)
        
        result = await self.session.execute(query)
        return list(result.scalars().all())
```
